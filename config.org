* Installation

** Personal

Private values live in `secrets.el`, which is gitignored. On a fresh machine the file is created with template values, so startup never fails on it — replace the placeholders with real values afterwards.

#+BEGIN_SRC emacs-lisp
  (let ((secrets-file (concat user-emacs-directory "secrets.el")))
    (unless (file-exists-p secrets-file)
      (with-temp-file secrets-file
        (insert ";;; secrets.el --- private values loaded by config.org -*- lexical-binding: t; -*-\n\n"
                ";; Replace the template values with your own.\n"
                "(setq full-name \"Your Name\"\n"
                "      email-address \"you@example.com\")\n"))
      (message "Created %s with template values -- please edit it" secrets-file))
    (load secrets-file))
  (setq user-full-name full-name
        user-mail-address email-address)
#+END_SRC

** Paths

Define the base directory for all org files — the single machine-specific data root; every other org path derives from it. Resolution is by existence, not by OS: take the first candidate directory that exists (iCloud on a synced Mac, `~/org/` elsewhere), and on a fresh machine create `~/org/` so startup always succeeds.

#+BEGIN_SRC emacs-lisp
  (setq org-directory
        (or (seq-find #'file-directory-p
                      (list (expand-file-name "~/Library/Mobile Documents/com~apple~CloudDocs/org/")
                            (expand-file-name "~/org/")))
            (let ((dir (expand-file-name "~/org/")))
              (make-directory dir t)
              (message "Created %s as org-directory" dir)
              dir)))
#+END_SRC

** Don't minimize frames

`C-z` iconifies GUI frames (to the dock on macOS, the taskbar on Linux) — never wanted. Disabling `suspend-frame` also covers its `C-x C-z` binding with a confirmation prompt. On a tty, `M-x suspend-emacs` remains available for shell job control.

#+BEGIN_SRC emacs-lisp
  (global-unset-key (kbd "C-z"))
  (put 'suspend-frame 'disabled t)
#+END_SRC

** Resize frames pixelwise

By default Emacs resizes frames in whole character-cell steps, which leaves gaps around a maximized window (visible between the frame and the macOS menu bar). Resize by pixels instead.

#+BEGIN_SRC emacs-lisp
  (setq frame-resize-pixelwise t)
#+END_SRC

** Delete by moving to trash

Deleted files go to the system trash instead of vanishing permanently — on every OS. macOS' trash folder is set in [[*Mac defaults]]; elsewhere Emacs follows the freedesktop.org trash specification.

#+BEGIN_SRC emacs-lisp

#+END_SRC

* Mac defaults
#+BEGIN_SRC emacs-lisp
  (when (string-equal system-type "darwin")
    ;; macOS' own trash folder; other systems follow the
    ;; freedesktop.org spec when this stays nil
    (setq trash-directory "~/.Trash")

    ;; Don't make new frames when opening a new file with Emacs
    (setq ns-pop-up-frames nil)

    ;; set the Fn key as the hyper key
    (setq ns-function-modifier 'hyper)

    ;; Use Command-` to switch between Emacs windows (not frames)
    (bind-key "s-`" 'other-window)

    ;; Use Command-Shift-` to switch Emacs windows in reverse
    (bind-key "s-~" (lambda () (interactive) (other-window -1)))

    ;; Because of the keybindings above, set one for `other-frame'
    (bind-key "s-1" 'other-frame)

    ;; Fullscreen!
    (setq ns-use-native-fullscreen nil) ; Not Lion style

    ;; Non-native fullscreen claims the display's full rect, including the band
    ;; macOS reserves below a MacBook Pro's camera housing -- cropping the top
    ;; line of text. Fill the monitor by hand instead, offset by that band's
    ;; height. It is zero on displays without a housing, where the frame falls
    ;; back to real fullscreen.
    (defvar my/notched-displays '((301 195))
      "`mm-size' values of displays whose top edge a camera housing covers.")

    (defvar my/fullscreen-restore nil
      "Alist mapping a frame to the geometry it had before pseudo-fullscreen.")

    (defun my/notch-height (frame)
      "Pixels at the top of FRAME's monitor that a camera housing covers."
      (if (member (frame-monitor-attribute 'mm-size frame) my/notched-displays)
          (- (nth 1 (frame-monitor-attribute 'workarea frame))
             (nth 1 (frame-monitor-attribute 'geometry frame)))
        0))

    (defun my/toggle-frame-fullscreen ()
      "Toggle fullscreen on the selected frame, clear of a camera housing."
      (interactive)
      (let* ((frame (selected-frame))
             (saved (assq frame my/fullscreen-restore))
             (notch (my/notch-height frame)))
        (cond
         (saved
          (pcase-let ((`(,x ,y ,w ,h) (cdr saved)))
            (modify-frame-parameters frame `((left + ,x) (top + ,y)))
            (set-frame-size frame w h t))
          (setq my/fullscreen-restore
                (assq-delete-all frame my/fullscreen-restore)))
         ((zerop notch)
          (toggle-frame-fullscreen))
         (t
          (set-frame-parameter frame 'fullscreen nil)
          (push (cons frame (list (car (frame-position frame))
                                  (cdr (frame-position frame))
                                  (frame-text-width frame)
                                  (frame-text-height frame)))
                my/fullscreen-restore)
          (pcase-let ((`(,x ,y ,w ,h) (frame-monitor-attribute 'geometry frame)))
            (modify-frame-parameters frame `((left + ,x) (top + ,(+ y notch))))
            (set-frame-size frame
                            (- w (- (frame-outer-width frame)
                                    (frame-text-width frame)))
                            (- h notch (- (frame-outer-height frame)
                                          (frame-text-height frame)))
                            t))))))

    (bind-key "<s-return>" 'my/toggle-frame-fullscreen)

    ;; buffer switching
    (bind-key "s-{" 'previous-buffer)
    (bind-key "s-}" 'next-buffer)

    ;; Not going to use this command
    (put 'ns-print-buffer 'disabled t))
#+END_SRC

* EasyPG encryption
#+BEGIN_SRC emacs-lisp
  (use-package epa-file
    :ensure nil
    :config
    (setq
     epa-file-encrypt-to email-address
     epa-file-select-keys 'silent
     epa-file-cache-passphrase-for-symmetric-encryption t
     epg-pinentry-mode 'loopback
     epg-gpg-program "gpg"))
#+END_SRC

Defining `epg-pinentry-mode 'loopback` enables the passphrase entering through minibuffer as pointed out by [[https://colinxy.github.io/software-installation/2016/09/24/emacs25-easypg-issue.html][colinxy]].

To encrypt any file with a public key associated for a receiver address, add the following line to the beginning of the file.

#+BEGIN_SRC txt
;; -*- epa-file-encrypt-to: ("me@domain.com") -*-
#+END_SRC

In case of an `.org` file use.

#+BEGIN_SRC txt
# -*- mode:org; epa-file-encrypt-to: ("me@domain.com") -*-
#+END_SRC

The passphrase will be cached depending on the configuration in `~/.gnupg/gpg-agent.conf`.

#+BEGIN_SRC txt
default-cache-ttl 600
max-cache-ttl 7200
#+END_SRC
* Fonts

Store the [[https://berkeleygraphics.com/typefaces/berkeley-mono/][Berkeley Mono]] Typeface in size 16 as symbol to use for any typeface decisions.

#+begin_src emacs-lisp
  (setq
   global-font "Berkeley Mono-16")
#+end_src

Set the overall font as general frame font and explicitly use it in orgmode's blocks and tables (`fixed-pitch`). Guarded by `find-font` so startup survives when the font is not installed. Under a daemon start no display exists during init and `find-font` cannot see any fonts, so the application is deferred to the first GUI frame; from there `set-frame-font` with `FRAMES` = t covers all future frames and the hook removes itself.

#+BEGIN_SRC emacs-lisp
  (defun global-font-apply ()
    "Apply `global-font' frame-wide once a graphical display can verify it."
    (when (and (display-graphic-p)
               (find-font (font-spec :name global-font)))
      (set-frame-font global-font nil t)
      (set-face-attribute 'fixed-pitch nil :font global-font)
      (remove-hook 'server-after-make-frame-hook #'global-font-apply)))

  (if (daemonp)
      (add-hook 'server-after-make-frame-hook #'global-font-apply)
    (global-font-apply))
#+END_SRC

`M-x locate` searches the system's file index. macOS has no `locate` database, so use Spotlight (`mdfind`) there; every other system keeps Emacs' default `locate` command (provided by e.g. mlocate/plocate on Linux).

#+begin_src emacs-lisp
  (when (and (eq system-type 'darwin)
             (executable-find "mdfind"))
    (setq locate-command "mdfind"))
#+end_src

* Minibuffer completion

The vertico family replaces ivy, counsel, swiper and amx. It builds on the standard `completing-read` API, so every package — magit, projectile, org, whatever comes next — gets the same UI without per-package glue commands or settings.

** Vertico

Vertical minibuffer UI. `savehist-mode` persists minibuffer histories across sessions; vertico sorts by recency from it, which covers the command ranking amx used to provide for `M-x`.

#+begin_src emacs-lisp
  (use-package vertico
    :ensure t
    :pin elpa-gnu
    :init
    (setq enable-recursive-minibuffers t)
    (savehist-mode 1)
    (vertico-mode 1))
#+end_src

Directory-wise editing in file prompts: `RET` enters a directory instead of finishing, `DEL` deletes whole path components — the old `ivy-alt-done` feel.

#+begin_src emacs-lisp
  (use-package vertico-directory
    :ensure nil ;; ships with vertico
    :after vertico
    :bind (:map vertico-map
                ("RET" . vertico-directory-enter)
                ("DEL" . vertico-directory-delete-char)
                ("M-DEL" . vertico-directory-delete-word)))
#+end_src

Resume the last minibuffer session, replacing `ivy-resume` on its old binding.

#+begin_src emacs-lisp
  (use-package vertico-repeat
    :ensure nil ;; ships with vertico
    :after vertico
    :hook (minibuffer-setup . vertico-repeat-save)
    :bind ("C-c C-r" . vertico-repeat))
#+end_src

* Exec-path-from-shell
#+BEGIN_SRC emacs-lisp
  (use-package exec-path-from-shell
    :ensure t
    :pin melpa-stable)

  ;; Besides PATH, import NIX_PROFILES: nix-patched tools rely on it,
  ;; e.g. aspell locates its dictionaries through it. GUI-launched
  ;; Emacs doesn't inherit it from launchd.
  (setq exec-path-from-shell-variables '("PATH" "MANPATH" "NIX_PROFILES"))

  ;; A non-interactive login shell suffices for reading environment
  ;; variables. The default adds -i for some shells (zsh, bash), whose

  (setq exec-path-from-shell-arguments '("-l"))

  ;; pgtk covers Wayland builds; daemonp covers daemon starts, where
  ;; window-system is nil during init.
  (when (or (memq window-system '(mac ns x pgtk))
            (daemonp))
    (exec-path-from-shell-initialize))

* Spell checking

Nix installs the dictionaries into the profile, but aspell resolves an explicitly named dictionary (ispell calls it with `-d en_US`) only against its built-in `dict-dir`, which points into aspell's own store path. Point it at the profile's dictionary directory instead — a stable symlink that survives store garbage collection, unlike hardcoded `/nix/store/...` hashes. Guarded by existence so non-nix machines are untouched, and an `ASPELL_CONF` provided by the environment always wins.

#+begin_src emacs-lisp
  (let ((dict-dir (expand-file-name "~/.nix-profile/lib/aspell")))
    (when (and (file-directory-p dict-dir)
               (not (getenv "ASPELL_CONF")))
      (setenv "ASPELL_CONF" (concat "dict-dir " dict-dir))))
#+end_src

On-the-fly spellchecking with [[https://github.com/minad/jinx][jinx]] where libenchant is available: it checks only the visible text through a native module, which makes it much faster than flyspell, and enchant's aspell backend reads the same dictionaries (honoring `ASPELL_CONF` above). The module compiles itself on first activation (needs a C compiler). Install the dependencies with:

#+begin_src shell
  nix profile install nixpkgs#enchant nixpkgs#pkg-config

* Verb

HTTP requests from org files, replacing the unmaintained restclient and ob-http: headings tagged `:verb:` hold request specs (method, URL, headers, body), and responses arrive pretty-printed in their own buffer. Request chaining and variables are covered in the [[https://github.com/federicotdn/verb][manual]].

#+BEGIN_SRC emacs-lisp
  (use-package verb
    :ensure t
    :pin melpa-stable
    :after org
    :config
    (define-key org-mode-map (kbd "C-c C-r") verb-command-map))
#+END_SRC

** Keys

| Binding       | Description                            |
|---------------+----------------------------------------|
| `C-c C-r C-r` | Send request at point                  |
| `C-c C-r C-s` | Send request, other window, stay       |
| `C-c C-r C-f` | Send request, show response body only  |

* Tree-sitter

Emacs 31 activates tree-sitter major modes through `treesit-enabled-modes` and installs missing grammars on demand (`treesit-auto-install-grammar` defaults to =ask=, prompting on first use; needs git and a C compiler). The source list below tells it where to fetch grammars from.

#+begin_src emacs-lisp
  (when (and (fboundp 'treesit-available-p)
             (treesit-available-p))
    (setq treesit-language-source-alist
          '((typescript "https://github.com/tree-sitter/tree-sitter-typescript" "master" "typescript/src")
            (tsx "https://github.com/tree-sitter/tree-sitter-typescript" "master" "tsx/src")
            (yaml "https://github.com/ikatyang/tree-sitter-yaml")
            (dockerfile "https://github.com/camdencheek/tree-sitter-dockerfile")))
    ;; `setopt', not `setq': the option's :set function performs the
    ;; major-mode remapping.
    (setopt treesit-enabled-modes
            '(typescript-ts-mode tsx-ts-mode yaml-ts-mode dockerfile-ts-mode))
    ;; .tsx is owned by web-mode, which must not be remapped wholesale
    ;; (it also serves .html); claim the extension directly instead.
    (add-to-list 'auto-mode-alist '("\\.tsx\\'" . tsx-ts-mode)))
#+end_src
* Web mode
For mixed-content templates and JSX/TSX. Plain `.css` files open in the built-in `css-mode` instead, where the eglot hook (see [[*Langauge server (Eglot)]]) provides LSP support.
#+begin_src emacs-lisp
  (use-package web-mode
    :ensure t
    :pin melpa-stable
    :mode
    ("\\.html\\'"
     "\\.tsx\\'")
    :config
    (setq
     web-mode-markup-indent-offset 2
     web-mode-css-indent-offset 2
     web-mode-code-indent-offset 2
     web-mode-enable-current-element-highlight t))
#+end_src
* Which key
Which key displays the key bindings following the currently entered incomplete command in a popup. Built into Emacs since version 30.
#+begin_src emacs-lisp
  (use-package which-key
    :ensure nil
    :diminish which-key-mode
    :hook (after-init . which-key-mode)
    :config
    (setq which-key-idle-delay 1))
#+end_src

* Writegood
Write clearly, easy to read, consice with the help of [[http://bnbeckwith.com/code/writegood-mode.html][writegood-mode]]. Inspired by [[https://matt.might.net/articles/shell-scripts-for-passive-voice-weasel-words-duplicates/][Matt Might's]] bash scripts.

Avoid common mistakes:
- Weasel Words
- Passive Voice
- Duplicated Words

#+begin_src emacs-lisp
  (use-package writegood-mode
    :ensure t
    :pin melpa
    :bind
    ("C-c C-g w" . writegood-mode)
    ("C-c C-g g" . writegood-grade-level)
    ("C-c C-g e" . writegood-reading-ease))
#+end_src

| Binding     | Description           |
|-------------+-----------------------|
| `C-c C-g w` | Toggle writegood-mode |
